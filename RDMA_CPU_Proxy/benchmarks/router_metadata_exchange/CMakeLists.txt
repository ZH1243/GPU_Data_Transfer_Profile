cmake_minimum_required(VERSION 3.18)
project(router_metadata_benchmark LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
option(ROUTER_METADATA_REQUIRE_RDMA "Fail configuration without libibverbs" OFF)
find_path(VERBS_INCLUDE_DIR infiniband/verbs.h)
find_library(VERBS_LIBRARY ibverbs)
set(HAVE_VERBS 0)
if(VERBS_INCLUDE_DIR AND VERBS_LIBRARY)
  set(HAVE_VERBS 1)
elseif(ROUTER_METADATA_REQUIRE_RDMA)
  message(FATAL_ERROR "RDMA requires libibverbs headers and library")
endif()
message(STATUS "Router metadata RDMA support: ${HAVE_VERBS}")
add_library(router_metadata_bench SHARED
  native_proxy.cpp
  ../../src/router_routing.cpp
  ../../src/protocol.cpp
  ../../src/logging.cpp ../../src/rdma_context.cpp ../../src/rdma_connection.cpp ../../src/config.cpp)
target_compile_features(router_metadata_bench PRIVATE cxx_std_17)
target_compile_definitions(router_metadata_bench PRIVATE RDMA_PROXY_HAVE_CUDA=0 RDMA_PROXY_HAVE_VERBS=${HAVE_VERBS})
target_include_directories(router_metadata_bench PRIVATE ../../include)
target_link_libraries(router_metadata_bench PRIVATE Threads::Threads)
target_compile_options(router_metadata_bench PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_binary_codec native_proxy.cpp ../../src/router_routing.cpp
    ../../src/protocol.cpp ../../src/logging.cpp ../../src/rdma_context.cpp ../../src/rdma_connection.cpp ../../src/config.cpp)
  target_compile_features(test_binary_codec PRIVATE cxx_std_17)
  target_compile_definitions(test_binary_codec PRIVATE RDMA_PROXY_HAVE_CUDA=0
    RDMA_PROXY_HAVE_VERBS=${HAVE_VERBS} ROUTER_METADATA_BENCH_CODEC_TEST=1)
  target_include_directories(test_binary_codec PRIVATE ../../include)
  target_link_libraries(test_binary_codec PRIVATE Threads::Threads)
  add_test(NAME binary_codec COMMAND test_binary_codec)
  set_tests_properties(binary_codec PROPERTIES TIMEOUT 15)
endif()

# shm_open lived in librt on older Linux/glibc installations.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(router_metadata_bench PRIVATE rt)
  if(BUILD_TESTING)
    target_link_libraries(test_binary_codec PRIVATE rt)
  endif()
endif()

if(HAVE_VERBS)
  target_include_directories(router_metadata_bench PRIVATE ${VERBS_INCLUDE_DIR})
  target_link_libraries(router_metadata_bench PRIVATE ${VERBS_LIBRARY})
  if(BUILD_TESTING)
    target_include_directories(test_binary_codec PRIVATE ${VERBS_INCLUDE_DIR})
    target_link_libraries(test_binary_codec PRIVATE ${VERBS_LIBRARY})
  endif()
endif()
