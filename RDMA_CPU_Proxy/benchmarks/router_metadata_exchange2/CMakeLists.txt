cmake_minimum_required(VERSION 3.18)
project(router_metadata_exchange2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(RME2_HOST_TESTS_ONLY "Build host metadata contract tests without CUDA" OFF)
option(RME2_REQUIRE_RDMA "Require libibverbs for two-node GPU-to-host exchange" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_two_node test_two_node.cpp ../../src/router_routing.cpp ../../src/logging.cpp)
  target_include_directories(test_two_node PRIVATE ../../include)
  target_compile_definitions(test_two_node PRIVATE RDMA_PROXY_HAVE_CUDA=0 RDMA_PROXY_HAVE_VERBS=0)
  target_compile_options(test_two_node PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME two_node COMMAND test_two_node)
  add_executable(test_payload_batch test_payload_batch.cpp)
  target_compile_options(test_payload_batch PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME payload_batch COMMAND test_payload_batch)
  add_executable(test_epoch_barrier test_epoch_barrier.cpp)
  target_compile_options(test_epoch_barrier PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME epoch_barrier COMMAND test_epoch_barrier)
  add_executable(test_readback_progress test_readback_progress.cpp)
  target_compile_options(test_readback_progress PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME readback_progress COMMAND test_readback_progress)
  add_executable(test_metadata_view test_metadata_view.cpp ../../src/router_routing.cpp ../../src/logging.cpp)
  target_include_directories(test_metadata_view PRIVATE ../../include)
  target_compile_definitions(test_metadata_view PRIVATE RDMA_PROXY_HAVE_CUDA=0 RDMA_PROXY_HAVE_VERBS=0)
  target_compile_options(test_metadata_view PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME metadata_view COMMAND test_metadata_view)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME topology COMMAND ${CMAKE_COMMAND} -E env PYTHONDONTWRITEBYTECODE=1
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_topology.py)
    add_test(NAME launch_dispatch COMMAND ${CMAKE_COMMAND} -E env PYTHONDONTWRITEBYTECODE=1
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_launch_dispatch.py)
    add_test(NAME reference COMMAND ${CMAKE_COMMAND} -E env
      RME2_REFERENCE_BIN=$<TARGET_FILE:test_metadata_view>
      PYTHONDONTWRITEBYTECODE=1
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_reference.py)
  endif()
endif()
if(NOT RME2_HOST_TESTS_ONLY)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "GPU benchmark requires remote Linux/CUDA; use RME2_HOST_TESTS_ONLY locally")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 90)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_package(Threads REQUIRED)
  find_path(VERBS_INCLUDE_DIR infiniband/verbs.h)
  find_library(VERBS_LIBRARY ibverbs)
  set(HAVE_VERBS 0)
  if(VERBS_INCLUDE_DIR AND VERBS_LIBRARY)
    set(HAVE_VERBS 1)
  elseif(RME2_REQUIRE_RDMA)
    message(FATAL_ERROR "Two-node exchange requires libibverbs headers and library")
  endif()
  message(STATUS "RME2 RDMA support: ${HAVE_VERBS}")
  add_library(router_metadata_exchange2 SHARED native_proxy.cpp exchange.cu
    ../../src/rdma_context.cpp ../../src/protocol.cpp
    ../../src/config.cpp ../../src/logging.cpp)
  target_include_directories(router_metadata_exchange2 PRIVATE ../../include)
  target_compile_definitions(router_metadata_exchange2 PRIVATE
    RDMA_PROXY_HAVE_CUDA=0 RDMA_PROXY_HAVE_VERBS=${HAVE_VERBS})
  if(HAVE_VERBS)
    target_include_directories(router_metadata_exchange2 PRIVATE ${VERBS_INCLUDE_DIR})
    target_link_libraries(router_metadata_exchange2 PRIVATE ${VERBS_LIBRARY})
  endif()
  set_target_properties(router_metadata_exchange2 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(router_metadata_exchange2 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
  target_link_libraries(router_metadata_exchange2 PRIVATE CUDA::cudart CUDA::cuda_driver Threads::Threads rt)
endif()
