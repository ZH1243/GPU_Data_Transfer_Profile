cmake_minimum_required(VERSION 3.20)
project(copy_engine_over_nvlink_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_GPU_BENCHMARK "Build Linux CUDA benchmark" ON)
include(CTest)
find_package(Threads REQUIRED)
if(BUILD_GPU_BENCHMARK)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "GPU benchmark requires Linux; use -DBUILD_GPU_BENCHMARK=OFF for host tests")
  endif()
  find_package(CUDAToolkit 12.8 REQUIRED)
  add_executable(nvlink_all_to_all src/main.cpp)
  target_link_libraries(nvlink_all_to_all PRIVATE CUDA::cudart Threads::Threads)
  target_compile_options(nvlink_all_to_all PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(host_tests tests/host_tests.cpp)
  target_include_directories(host_tests PRIVATE src)
  target_link_libraries(host_tests PRIVATE Threads::Threads)
  target_compile_options(host_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME host_tests COMMAND host_tests)
  set_tests_properties(host_tests PROPERTIES TIMEOUT 30)
endif()
